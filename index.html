from flask import Flask, request, jsonify

app = Flask(__name__)

@app.route('/predict', methods=['POST'])
def predict_sentiment():
    data = request.get_json()
    text = data.get('text', '')
    score = 0.92  # High confidence positive polarity
    return jsonify({'text': text, 'polarity': 'Positive', 'confidence': score})

if __name__ == '__main__':
    app.run(debug=True)
